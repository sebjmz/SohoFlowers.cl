let currentCompressedBase64 = null; function handleImageSelection(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const img = new Image(); img.onload = function() { const maxDim = 1000; let width = img.width; let height = img.height; if (width > height && width > maxDim) { height = Math.round((height * maxDim) / width); width = maxDim; } else if (height > maxDim) { width = Math.round((width * maxDim) / height); height = maxDim; } const canvas = document.createElement('canvas'); canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0, width, height); // Compresión en JPEG al 60% (~80-120 KB) currentCompressedBase64 = canvas.toDataURL('image/jpeg', 0.6); const preview = document.getElementById('preview'); if (preview) { preview.src = currentCompressedBase64; preview.classList.remove('hidden'); } }; img.src = e.target.result; }; reader.readAsDataURL(file); }
